async function ghRequest(path, method = 'GET', body = null) { const user = document.getElementById('gh-user').value.trim(); const repo = document.getElementById('gh-repo').value.trim(); const token = document.getElementById('gh-token').value.trim(); if (!token) throw new Error('GitHub Token required!'); const url = `https://api.github.com/repos/${user}/${repo}/contents/${path}`; const headers = { 'Authorization': `Bearer ${token}`, 'Accept': 'application/vnd.github.v3+json', 'Content-Type': 'application/json' }; const options = { method, headers }; if (body) options.body = JSON.stringify(body); const res = await fetch(url, options); if (res.status === 404 && method === 'GET') return null; if (!res.ok) { const err = await res.json(); throw new Error(err.message || 'GitHub API error'); } return await res.json(); }
